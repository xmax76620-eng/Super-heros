function esc(s) { return String(s == null ? '' : s).replace(/[&<>"']/g, c => ({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c])); }

function fileToDataUri(file) {
return new Promise((resolve, reject) => {
const reader = new FileReader();
reader.onload = e => resolve(e.target.result);
reader.onerror = () => reject(new Error('Erreur'));
reader.readAsDataURL(file);
});
}

function generateThumbnail(dataUri) {
return new Promise((resolve) => {
const img = new Image();
img.onload = () => {
try {
const canvas = document.createElement('canvas');
const ratio = img.height / img.width;
canvas.width = THUMBNAIL_MAX_WIDTH;
canvas.height = Math.round(THUMBNAIL_MAX_WIDTH * ratio);
const ctx = canvas.getContext('2d');
ctx.drawImage(img, 0, 0, canvas.width, canvas.height);
resolve(canvas.toDataURL('image/jpeg', 0.7));
} catch (e) { resolve(dataUri); }
};
img.onerror = () => resolve(dataUri);
img.src = dataUri;
});
}

async function processOneFile(file) {
if (!file.type.startsWith('image/')) return null;
if (file.size > MAX_IMAGE_SIZE) return null;
try {
const dataUri = await fileToDataUri(file);
const thumbnail = await generateThumbnail(dataUri);
return { file, dataUri, thumbnail, id: Date.now() + Math.random() };
} catch (e) { return null; }
}

async function handleFiles(files) {
const imageFiles = Array.from(files).filter(f => f.type.startsWith('image/'));
if (!imageFiles.length) return;
state.isLoadingImages = true;
updateGenerateBtn();
setStatus('Chargement…');
for (let i = 0; i < imageFiles.length; i++) {
const result = await processOneFile(imageFiles[i]);
if (result) state.images.push(result);
renderImages();
}
state.isLoadingImages = false;
setStatus(null);
updateGenerateBtn();
}

function removeImage(id) {
state.images = state.images.filter(img => img.id !== id);
renderImages();
updateGenerateBtn();
}

function renderImages() {
const grid = document.getElementById('images-grid');
grid.innerHTML = state.images.map(img =>
'<div class="image-card"><img src="' + (img.thumbnail || img.dataUri) + '"><button class="remove-btn" data-remove="' + img.id + '">✕</button></div>'
).join('');
grid.querySelectorAll('[data-remove]').forEach(btn => {
btn.addEventListener('click', (e) => {
e.stopPropagation();
removeImage(parseFloat(btn.dataset.remove));
});
});
}

function renderHeroes() {
const container = document.getElementById('heroes-grid');
container.innerHTML = HEROES.map(h =>
'<div class="hero-card ' + (state.selectedHero === h.id ? 'selected' : '') + '" data-hero="' + h.id + '">' +
'<div class="hero-emoji">' + h.emoji + '</div>' +
'<div><div class="hero-name">' + esc(h.name) + '</div>' +
'<div class="hero-desc">' + esc(h.desc) + '</div></div></div>'
).join('');
container.querySelectorAll('[data-hero]').forEach(el => {
el.addEventListener('click', () => {
state.selectedHero = el.dataset.hero;
renderHeroes();
});
});
}

function buildPrompt() {
const hero = state.selectedHero ? HEROES.find(h => h.id === state.selectedHero) : null;
const parts = [];
parts.push('Animate this exact image as the starting frame. Preserve the subject identity, face, and composition.');
if (hero) parts.push(hero.prompt);
if (state.mainPrompt && state.mainPrompt.trim()) parts.push('Additional direction: ' + state.mainPrompt.trim());
parts.push('Portrait orientation 9:16 vertical, 24fps, no text, no watermarks.');
return parts.join('. ');
}

async function apiFetch(url, options, label) {
for (let attempt = 0; attempt < 7; attempt++) {
if (state.stopRequested) throw new Error('Arrêt demandé');
try {
const res = await fetch(url, options);
if (res.status === 429) { await sleep(30000); continue; }
if (res.status === 503) { await sleep(10000); continue; }
return res;
} catch (e) {
if (e.message === 'Arrêt demandé') throw e;
await sleep(5000);
}
}
return fetch(url, options);
}
